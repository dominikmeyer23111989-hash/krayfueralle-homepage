// Login direkt über die mitglieder-Tabelle mit Rechte-Prüfung loginForm.addEventListener('submit', async function(e) { e.preventDefault(); loginError.classList.add('hidden'); const email = document.getElementById('email').value; const passwordInput = document.getElementById('password').value; // Hier nutzen wir das Passwort-Feld als Identifikation/PIN oder Sicherheitsantwort try { // Wir suchen in der mitglieder-Tabelle nach der E-Mail const { data: memberData, error } = await supabaseClient .from('mitglieder') .select('*') .eq('email', email) .single(); if (error || !memberData) { throw new Error("E-Mail-Adresse nicht in der Mitgliederdatenbank gefunden."); } // Prüfen ob Account gesperrt ist if (memberData.ist_gesperrt) { throw new Error("Dieser Mitglieds-Account ist gesperrt."); } // Prüfen ob die Person Admin-/Redaktionsrechte hat (z.B. hat_inventar_rechte oder Rolle) if (!memberData.hat_inventar_rechte && memberData.rolle !== 'Vorstand' && memberData.rolle !== 'Admin') { throw new Error("Keine Berechtigung für den Redaktions- und Admin-Bereich."); } // Optional: Abgleich mit der Sicherheitsantwort oder einer PIN (falls so gewünscht) // Wenn du ein echtes Passwort hinterlegen möchtest, könnten wir dafür auch die sicherheitsantwort nutzen: if (memberData.sicherheitsantwort && memberData.sicherheitsantwort !== passwordInput) { throw new Error("Falsches Passwort / Falsche Sicherheitsantwort."); } // Login erfolgreich: Sitzung im Browser speichern localStorage.setItem('kray_admin_logged_in', 'true'); localStorage.setItem('kray_admin_email', memberData.email); localStorage.setItem('kray_admin_name', memberData.vorname + ' ' + (memberData.nachname || '')); checkSession(); } catch (err) { loginError.textContent = 'Login fehlgeschlagen: ' + err.message; loginError.classList.remove('hidden'); } }); // Session-Prüfung über den lokalen Status function checkSession() { const isLoggedIn = localStorage.getItem('kray_admin_logged_in'); if (isLoggedIn === 'true') { loginCard.classList.add('hidden'); adminCard.classList.remove('hidden'); loadManageNews(); loadImpressumAdmin(); } else { loginCard.classList.remove('hidden'); adminCard.classList.add('hidden'); } } // Logout verarbeiten logoutBtn.addEventListener('click', function() { localStorage.removeItem('kray_admin_logged_in'); localStorage.removeItem('kray_admin_email'); localStorage.removeItem('kray_admin_name'); checkSession(); });
